class RIYAAssistant:
    """
    Main RIYA application interface.

    Public interface:

        assistant = RIYAAssistant()
        assistant.process("...")
        assistant.close()

    Architecture:

        User
          ↓
        Memory
          ↓
        Deterministic Router
          ↓
        Semantic Tool Retrieval
          ↓
        AI Planner
          ↓
        Security
          ↓
        Tool Execution
          ↓
        Verification
          ↓
        RIYA Response
    """

    def __init__(self):

        # =====================================================
        # RIYA AGENT
        # =====================================================

        self.agent = RiyaAgent()

        # =====================================================
        # LOCAL SEMANTIC ENGINE
        # =====================================================

        self.semantic = SemanticToolEngine(
            self.agent.tools,
            top_k=8,
        )

        # =====================================================
        # LOCAL AI
        # =====================================================

        self.ai = LocalLlamaRuntime(
            context_size=4096,
            max_tokens=256,
        )

        # =====================================================
        # MODEL
        # =====================================================

        self.model_path = self._find_model()

        self.ai.load_model(
            str(self.model_path)
        )

        # =====================================================
        # CORE COMPONENTS
        # =====================================================

        self.intent_detector = IntentDetector()

        self.command_executor = CommandExecutor(
            self.agent
        )

        self.memory = MemoryService()

        # =====================================================
        # AI PLANNER + SEMANTIC ENGINE
        # =====================================================

        self.planner = AgentPlanner(
            self.ai,
            self.agent.tools,
            semantic_engine=self.semantic,
        )

        # =====================================================
        # CONVERSATION CONTEXT
        # =====================================================

        self.context = ConversationContext(
            max_messages=12
        )

        # =====================================================
        # PENDING CONFIRMATION
        # =====================================================

        self.pending_action = None

    def _find_model(self):

        from pathlib import Path

        candidates = [

            Path.home()
            / "Projects"

        ]

        # -----------------------------------------------------
        # Direct model file

        for candidate in candidates:

            if candidate.is_file():

                return candidate

        # -----------------------------------------------------
        # Search Hugging Face snapshots
        # -----------------------------------------------------

        hf_snapshots = candidates[-1]

        if hf_snapshots.exists():

            for snapshot in hf_snapshots.iterdir():

                if not snapshot.is_dir():
                    continue

                model_file = (
                    snapshot

    # =========================================================
    # MAIN PROCESSING
    # =========================================================

    def process(
        self,
        user_input: str,
    ) -> str:

        user_input = user_input.strip()

        # -----------------------------------------------------
        # Empty input
        # -----------------------------------------------------

        if not user_input:

            return (
                "Please tell me what you'd like me to do."
            )

        # -----------------------------------------------------
        # Pending confirmation
        # -----------------------------------------------------

        if self.pending_action is not None:

            return self._handle_confirmation(
                user_input
            )

        # =====================================================
        # MEMORY SYSTEM
        # =====================================================

        memory_intent = self.memory.detect(
            user_input
        )

        # -----------------------------------------------------
        # Recall
        # -----------------------------------------------------

        if (
            memory_intent.type
            == MemoryIntentType.RECALL
        ):

            response = self.memory.execute_safe(
                memory_intent
            )

            response = response or (
                "I couldn't find that memory."
            )

            self._update_context(
                user_input,
                response,
            )

            return response

        # -----------------------------------------------------
        # List memories
        # -----------------------------------------------------

        if (
            memory_intent.type
            == MemoryIntentType.LIST
        ):

            response = self.memory.execute_safe(
                memory_intent
            )

            response = response or (
                "RIYA has no stored memories."
            )

            self._update_context(
                user_input,
                response,
            )

            return response

        # -----------------------------------------------------
        # Remember
        # -----------------------------------------------------

        if (
            memory_intent.type
            == MemoryIntentType.REMEMBER
        ):

            action = (
                self.memory.build_remember_action(
                    memory_intent
                )
            )

            self.pending_action = {
                "tool": action["tool"],
                "parameters": action["parameters"],
            }

            category = (
                action["parameters"]["category"]
            )

            key = (
                action["parameters"]["key"]
            )

            value = (
                action["parameters"]["value"]
            )

            response = (
                f"I'll remember '{value}' "
                f"under '{category}' as '{key}'. "
                f"Proceed?"
            )

            self._update_context(
                user_input,
                response,
            )

            return response

        # -----------------------------------------------------
        # Forget
        # -----------------------------------------------------

        if (
            memory_intent.type
            == MemoryIntentType.FORGET
        ):

            action = (
                self.memory.build_forget_action(
                    memory_intent
                )
            )

            self.pending_action = {
                "tool": action["tool"],
                "parameters": action["parameters"],
            }

            category = (
                action["parameters"]["category"]
            )

            key = (
                action["parameters"]["key"]
            )

            response = (
                f"I'll forget the memory "
                f"'{key}' under '{category}'. "
                f"Proceed?"
            )

            self._update_context(
                user_input,
                response,
            )

            return response

        # =====================================================
        # DETERMINISTIC COMMAND ROUTER
        # =====================================================

        intent = self.intent_detector.detect(
            user_input
        )

        # -----------------------------------------------------
        # Exit
        # -----------------------------------------------------

        if intent.type == IntentType.EXIT:

            return "Goodbye."

        # -----------------------------------------------------
        # Simple deterministic command
        # -----------------------------------------------------

        if intent.type != IntentType.CHAT:

            try:

                result = (
                    self.command_executor.execute(
                        intent
                    )
                )

                response = self._format_result(
                    result
                )

                self._update_context(
                    user_input,
                    response,
                )

                return response

            except Exception as exc:

                response = (
                    "I couldn't complete that action: "
                    f"{exc}"
                )

                self._update_context(
                    user_input,
                    response,
                )

                return response

        # =====================================================
        # SEMANTIC + AI PLANNER
        # =====================================================

        try:

            plan = self.planner.create_plan(
                user_input
            )

        except Exception:

            plan = None

        # =====================================================
        # NORMAL CONVERSATION
        # =====================================================

        if plan is None or not plan.steps:

            response = self.ai.generate(
                user_input,
                conversation_context=(
                    self.context.format_for_prompt()
                ),
            )

            self._update_context(
                user_input,
                response,
            )

            return response

        # =====================================================
        # MULTI-STEP EXECUTION
        # =====================================================

        results = []

        for step in plan.steps:

            try:

                result = self.agent.execute(
                    step.action,
                    **step.parameters,
                )

            except Exception as exc:

                response = (
                    "I couldn't complete the full task. "
                    f"The operation "
                    f"'{step.action}' failed: {exc}"
                )

                self._update_context(
                    user_input,
                    response,
                )

                return response

            # -------------------------------------------------
            # Confirmation required
            # -------------------------------------------------

            if (
                isinstance(result, dict)
                and result.get("status")
                == "confirmation_required"
            ):

                self.pending_action = {
                    "tool": result.get(
                        "tool",
                        step.action,
                    ),
                    "parameters": result.get(
                        "parameters",
                        step.parameters,
                    ),
                }

                response = (
                    f"{result.get('reason', 'This action requires confirmation.')} "
                    "Proceed?"
                )

                self._update_context(
                    user_input,
                    response,
                )

                return response

            # -------------------------------------------------
            # Execution failure
            # -------------------------------------------------

            if (
                isinstance(result, dict)
                and result.get("status")
                in {
                    "failed",
                    "verification_failed",
                }
            ):

                reason = result.get(
                    "reason",
                    result.get(
                        "error",
                        "Unknown failure.",
                    ),
                )

                response = (
                    f"I couldn't complete "
                    f"'{step.description or step.action}'. "
                    f"{reason}"
                )

                self._update_context(
                    user_input,
                    response,
                )

                return response

            # -------------------------------------------------
            # Successful result
            # -------------------------------------------------

            results.append(
                {
                    "action": step.action,
                    "description": step.description,
                    "result": result,
                }
            )

        # =====================================================
        # FINAL RESULT
        # =====================================================

        response = self._summarize_results(
            results
        )

        self._update_context(
            user_input,
            response,
        )

        return response

    # =========================================================
    # CONFIRMATION
    # =========================================================

    def _handle_confirmation(
        self,
        user_input: str,
    ) -> str:

        normalized = (
            user_input.lower()
            .strip()
            .rstrip(".!?")
        )

        yes_words = {
            "yes",
            "y",
            "yeah",
            "yep",
            "sure",
            "ok",
            "okay",
            "confirm",
            "do it",
            "proceed",
            "go ahead",
            "please proceed",
            "continue",
        }

        no_words = {
            "no",
            "n",
            "nope",
            "cancel",
            "stop",
            "don't",
            "dont",
        }

        # -----------------------------------------------------
        # Cancel
        # -----------------------------------------------------

        if normalized in no_words:

            self.pending_action = None

            response = (
                "Okay, I cancelled that action."
            )

            self._update_context(
                user_input,
                response,
            )

            return response

        # -----------------------------------------------------
        # Invalid confirmation
        # -----------------------------------------------------

        if normalized not in yes_words:

            return (
                "Please say yes to proceed "
                "or no to cancel."
            )

        # -----------------------------------------------------
        # Execute confirmed action
        # -----------------------------------------------------

        action = self.pending_action

        self.pending_action = None

        try:

            result = self.agent.execute(
                action["tool"],
                confirm=True,
                **action["parameters"],
            )

            response = self._format_result(
                result
            )

            self._update_context(
                user_input,
                response,
            )

            return response

        except Exception as exc:

            response = (
                "I couldn't complete the "
                f"confirmed action: {exc}"
            )

            self._update_context(
                user_input,
                response,
            )

            return response

    # =========================================================
    # RESULT FORMATTING
    # =========================================================

    def _format_result(
        self,
        result,
    ) -> str:

        # -----------------------------------------------------
        # Plain string
        # -----------------------------------------------------

        if isinstance(result, str):

            return result.strip()

        # -----------------------------------------------------
        # Structured result
        # -----------------------------------------------------

        if isinstance(result, dict):

            status = result.get(
                "status"
            )

            # -------------------------------------------------
            # Confirmation
            # -------------------------------------------------

            if status == "confirmation_required":

                self.pending_action = {
                    "tool": result.get(
                        "tool"
                    ),
                    "parameters": result.get(
                        "parameters",
                        {},
                    ),
                }

                return (
                    f"{result.get('reason', 'Confirmation required.')} "
                    "Proceed?"
                )

            # -------------------------------------------------
            # Failed
            # -------------------------------------------------

            if status == "failed":

                return (
                    "I couldn't complete the action. "
                    f"{result.get('error', 'Unknown error.')}"
                )

            # -------------------------------------------------
            # Verification failed
            # -------------------------------------------------

            if status == "verification_failed":

                return (
                    "The action was attempted, "
                    "but I could not verify that it "
                    "completed successfully. "
                    f"{result.get('reason', '')}"
                ).strip()

            # -------------------------------------------------
            # Successful
            # -------------------------------------------------

            if status == "success":

                actual_result = result.get(
                    "result"
                )

                if isinstance(
                    actual_result,
                    str,
                ):

                    return actual_result.strip()

                if actual_result is not None:

                    return str(
                        actual_result
                    )

                verification = result.get(
                    "verification"
                )

                if isinstance(
                    verification,
                    dict,
                ):

                    reason = verification.get(
                        "reason"
                    )

                    if reason:

                        return str(
                            reason
                        )

                return (
                    "The action completed successfully."
                )

            # -------------------------------------------------
            # Unknown dictionary
            # -------------------------------------------------

            return str(result)

        return str(result)

    # =========================================================
    # MULTI-STEP SUMMARY
    # =========================================================

    def _summarize_results(
        self,
        results,
    ) -> str:

        if not results:

            return (
                "The task completed."
            )

        successful = []

        for item in results:

            raw_result = item.get(
                "result"
            )

            # -------------------------------------------------
            # Structured ToolExecutor result
            # -------------------------------------------------

            if isinstance(
                raw_result,
                dict,
            ):

                status = raw_result.get(
                    "status"
                )

                if status == "success":

                    actual_result = (
                        raw_result.get(
                            "result"
                        )
                    )

                    if actual_result is not None:

                        text = str(
                            actual_result
                        ).strip()

                        if text:

                            successful.append(
                                text
                            )

                        continue

                    verification = (
                        raw_result.get(
                            "verification"
                        )
                    )

                    if isinstance(
                        verification,
                        dict,
                    ):

                        reason = (
                            verification.get(
                                "reason"
                            )
                        )

                        if reason:

                            successful.append(
                                str(
                                    reason
                                ).strip()
                            )

                continue

            # -------------------------------------------------
            # Plain string result
            # -------------------------------------------------

            if isinstance(
                raw_result,
                str,
            ):

                cleaned = (
                    raw_result.strip()
                )

                if cleaned:

                    successful.append(
                        cleaned
                    )

        # -----------------------------------------------------
        # No readable result
        # -----------------------------------------------------

        if not successful:

            return (
                "Done. The requested actions "
                "were completed and verified."
            )

        # -----------------------------------------------------
        # One result
        # -----------------------------------------------------

        if len(successful) == 1:

            return successful[0]

        # -----------------------------------------------------
        # Multiple results
        # -----------------------------------------------------

        lines = [
            "Done. I completed the requested actions:"
        ]

        for result in successful:

            lines.append(
                f"• {result}"
            )

        return "\n".join(
            lines
        )

    # =========================================================
    # CONVERSATION CONTEXT
    # =========================================================

    def _update_context(
        self,
        user_input: str,
        response: str,
    ) -> None:

        self.context.add_user(
            user_input
        )

        if response:

            self.context.add_assistant(
                response
            )

    # =========================================================
    # SHUTDOWN
    # =========================================================

    def close(self):

        if self.ai is not None:

            try:

                self.ai.unload_model()

            except Exception:

                pass

        )

        if decision.tool is None:
            return None

        return self.execute(
            decision.tool,
            **decision.parameters,
        )

from core.agent.intent import Intent, IntentType


class CommandRouter:
    """Convert user text into a RIYA intent."""

    def route(self, text: str) -> Intent:
        text = text.strip()
        lower = text.lower()

        # Exit
        if lower in {"exit", "quit", "stop", "goodbye"}:
            return Intent(IntentType.EXIT)

        # Open application
        if lower.startswith("open "):
            app_name = text[5:].strip()

            if app_name:
                return Intent(
                    IntentType.OPEN_APPLICATION,
                    {"app_name": app_name},
                )

        # Close application
        if lower.startswith("close "):
            app_name = text[6:].strip()

            if app_name:
                return Intent(
                    IntentType.CLOSE_APPLICATION,
                    {"app_name": app_name},
                )

        # Create folder
        if lower.startswith("create folder "):
            folder_name = text[14:].strip()

            if folder_name:
                return Intent(
                    IntentType.CREATE_FOLDER,
                    {"folder_name": folder_name},
                )

        # List files
        if lower in {
            "list files",
            "show files",
            "show my files",
        }:
            return Intent(IntentType.LIST_FILES)

        # Screenshot
        if lower in {
            "screenshot",
            "take screenshot",
            "take a screenshot",
            "capture screen",
        }:
            return Intent(IntentType.SCREENSHOT)

        # Current time
        if lower in {
            "time",
            "what time is it",
            "what is the time",
            "current time",
        }:
            return Intent(IntentType.CURRENT_TIME)

        # Everything else → AI conversation
        return Intent(IntentType.CHAT)

class ConversationContext:
    """
    Short-term conversation context.

    This data lives only for the current RIYA process/session.
    It is NOT written to the persistent memory database.
    """

    def __init__(self, max_messages: int = 12):
        if max_messages < 2:
            raise ValueError(
                "max_messages must be at least 2."
            )

        self.max_messages = max_messages
        self._messages: list[ConversationMessage] = []

    def add_user(self, content: str) -> None:
        self._add(
            role="user",
            content=content,
        )

    def add_assistant(self, content: str) -> None:
        self._add(
            role="assistant",
            content=content,
        )

    def _add(
        self,
        role: str,
        content: str,
    ) -> None:

        content = content.strip()

        if not content:
            return

        self._messages.append(
            ConversationMessage(
                role=role,
                content=content,
            )
        )

        self._trim()

    def _trim(self) -> None:
        if len(self._messages) <= self.max_messages:
            return

        self._messages = self._messages[
            -self.max_messages:
        ]

    def messages(self) -> list[ConversationMessage]:
        return list(self._messages)

    def clear(self) -> None:
        self._messages.clear()

    def __len__(self) -> int:
        return len(self._messages)

    def format_for_prompt(self) -> str:
        if not self._messages:
            return ""

        lines = []

        for message in self._messages:
            if message.role == "user":
                speaker = "User"
            else:
                speaker = "RIYA"

            lines.append(
                f"{speaker}: {message.content}"
            )

        return "\n".join(lines)

from core.agent.intent import IntentType


class CommandExecutor:
    """
    Compatibility adapter for RIYA's command router.

    All real actions are delegated to RiyaAgent so that execution
    passes through the central tool registry and security layer.
    """

    def __init__(self, agent=None):
        self.agent = agent

    def set_agent(self, agent):
        self.agent = agent

    def execute(self, intent):
        if self.agent is None:
            raise RuntimeError(
                "CommandExecutor is not connected to RiyaAgent."
            )

        intent_type = intent.type

        # Some older Intent implementations store arguments
        # differently. Normalize them here.
        parameters = getattr(intent, "parameters", None)

        if parameters is None:
            parameters = {}

            # Compatibility with Intent objects that use
            # dedicated attributes instead of parameters.
            if hasattr(intent, "app_name"):
                parameters["app_name"] = intent.app_name

            if hasattr(intent, "path"):
                parameters["path"] = intent.path

            if hasattr(intent, "directory"):
                parameters["directory"] = intent.directory

        if intent_type == IntentType.OPEN_APPLICATION:
            app_name = parameters.get("app_name")

            if not app_name:
                raise ValueError(
                    "No application name was provided."
                )

            return self.agent.execute(
                "open_application",
                app_name=app_name,
            )

        if intent_type == IntentType.CLOSE_APPLICATION:
            app_name = parameters.get("app_name")

            if not app_name:
                raise ValueError(
                    "No application name was provided."
                )

            return self.agent.execute(
                "close_application",
                app_name=app_name,
            )

        if intent_type == IntentType.CREATE_FOLDER:
            path = parameters.get("path")

            if not path:
                raise ValueError(
                    "No folder path was provided."
                )

            return self.agent.execute(
                "create_folder",
                path=path,
            )

        if intent_type == IntentType.LIST_FILES:
            path = parameters.get(
                "path",
                parameters.get("directory", "."),
            )

            return self.agent.execute(
                "list_files",
                path=path,
            )

        if intent_type == IntentType.SCREENSHOT:
            return self.agent.execute(
                "take_screenshot"
            )

        if intent_type == IntentType.CURRENT_TIME:
            return self.agent.execute(
                "current_time"
            )

        if intent_type == IntentType.EXIT:
            return "Goodbye."

        if intent_type == IntentType.CHAT:
            return None

        raise ValueError(
            f"Unsupported RIYA intent: {intent_type}"
        )

from enum import Enum


class IntentType(str, Enum):

    EXIT = "exit"


@dataclass
class Intent:

class IntentDetector:
    """
    Deterministic detector for simple, unambiguous commands.

    Natural-language commands that may contain multiple actions
    are deliberately routed to CHAT so RIYA's AI planner can
    decompose them into multiple tool actions.
    """

    # =========================================================
    # EXIT COMMANDS
    # =========================================================

    EXIT_COMMANDS = {
        "exit",
        "quit",
        "bye",
        "goodbye",
        "close riya",
        "stop riya",
    }

    # =========================================================
    # SCREENSHOT COMMANDS
    # =========================================================

    SCREENSHOT_COMMANDS = {
        "screenshot",
        "take screenshot",
        "take a screenshot",
        "capture screen",
        "capture the screen",
        "capture my screen",
        "take a screen capture",
        "capture a screenshot",
    }

    # =========================================================
    # CURRENT TIME COMMANDS
    # =========================================================

    CURRENT_TIME_COMMANDS = {
        "time",
        "current time",
        "what time is it",
        "what is the time",
        "tell me the time",
        "what's the time",
        "whats the time",
        "can you tell me the time",
    }

    # =========================================================
    # APPLICATION VERBS
    # =========================================================

    OPEN_VERBS = (
        "open ",
        "launch ",
        "start ",
        "run ",
    )

    CLOSE_VERBS = (
        "close ",
        "quit ",
        "exit ",
        "stop ",
    )

    # =========================================================
    # FOLDER COMMANDS
    # =========================================================

    CREATE_FOLDER_PREFIXES = (
        "create a folder called ",
        "create a folder named ",
        "create folder called ",
        "create folder named ",
        "make a folder called ",
        "make a folder named ",
        "make folder called ",
        "make folder named ",
    )

    # =========================================================
    # FILE LISTING COMMANDS
    # =========================================================

    LIST_FILES_PREFIXES = (
        "show files in ",
        "list files in ",
        "show me files in ",
        "list files from ",
        "show the files in ",
        "display files in ",
        "display the files in ",
    )

    # =========================================================
    # MAIN DETECTOR
    # =========================================================

    def detect(
        self,
        text: str,
    ) -> Intent:

        text = text.strip()

        if not text:
            return Intent(
                IntentType.CHAT
            )

        normalized = self._normalize(
            text
        )

        # -----------------------------------------------------
        # Exit
        # -----------------------------------------------------

        if normalized in self.EXIT_COMMANDS:

            return Intent(
                IntentType.EXIT
            )

        # -----------------------------------------------------
        # Multi-action request
        #
        # These must go to the AI planner.
        # -----------------------------------------------------

        if self._is_multi_action_request(
            normalized
        ):

            return Intent(
                IntentType.CHAT
            )

        # -----------------------------------------------------
        # Screenshot
        # -----------------------------------------------------

        if normalized in self.SCREENSHOT_COMMANDS:

            return Intent(
                IntentType.SCREENSHOT
            )

        # -----------------------------------------------------
        # Current time
        # -----------------------------------------------------

        if normalized in self.CURRENT_TIME_COMMANDS:

            return Intent(
                IntentType.CURRENT_TIME
            )

        # -----------------------------------------------------
        # Open application
        # -----------------------------------------------------

        app_name = self._extract_application_name(
            text,
            normalized,
            self.OPEN_VERBS,
        )

        if app_name:

            return Intent(
                IntentType.OPEN_APPLICATION,
                {
                    "app_name": app_name,
                },
            )

        # -----------------------------------------------------
        # Close application
        # -----------------------------------------------------

        app_name = self._extract_application_name(
            text,
            normalized,
            self.CLOSE_VERBS,
        )

        if app_name:

            return Intent(
                IntentType.CLOSE_APPLICATION,
                {
                    "app_name": app_name,
                },
            )

        # -----------------------------------------------------
        # Create folder
        # -----------------------------------------------------

        for prefix in self.CREATE_FOLDER_PREFIXES:

            if normalized.startswith(prefix):

                folder_name = text[
                    len(prefix):
                ].strip()

                if folder_name:

                    return Intent(
                        IntentType.CREATE_FOLDER,
                        {
                            "path": folder_name,
                        },
                    )

        # -----------------------------------------------------
        # List files
        # -----------------------------------------------------

        for prefix in self.LIST_FILES_PREFIXES:

            if normalized.startswith(prefix):

                path = text[
                    len(prefix):
                ].strip()

                if path:

                    return Intent(
                        IntentType.LIST_FILES,
                        {
                            "path": self._normalize_path_alias(
                                path
                            ),
                        },
                    )

        # -----------------------------------------------------
        # No deterministic command found
        # -----------------------------------------------------

        return Intent(
            IntentType.CHAT
        )

    # =========================================================
    # NORMALIZATION
    # =========================================================

    def _normalize(
        self,
        text: str,
    ) -> str:

        normalized = text.lower().strip()

        # Remove common punctuation at the end.
        normalized = normalized.rstrip(
            "!?.,"
        )

        # Collapse repeated whitespace.
        normalized = " ".join(
            normalized.split()
        )

        return normalized

    # =========================================================
    # APPLICATION NAME EXTRACTION
    # =========================================================

    def _extract_application_name(
        self,
        original_text: str,
        normalized_text: str,
        verbs,
    ) -> str | None:

        for verb in verbs:

            if not normalized_text.startswith(
                verb
            ):
                continue

            # Use the original text so application names
            # retain their original capitalization.

            app_name = original_text[
                len(verb):
            ].strip()

            app_name = app_name.rstrip(
                "!?.,"
            ).strip()

            if not app_name:
                return None

            # -------------------------------------------------
            # Natural-language suffixes
            #
            # Example:
            # "Open Safari for me"
            # → Safari
            # -------------------------------------------------

            suffixes = (
                " for me",
                " please",
                " for me please",
            )

            lowered_name = app_name.lower()

            for suffix in suffixes:

                if lowered_name.endswith(
                    suffix
                ):

                    app_name = app_name[
                        : -len(suffix)
                    ].strip()

                    break

            if app_name:
                return app_name

            return None

        return None

    # =========================================================
    # MULTI-ACTION DETECTION
    # =========================================================

    def _is_multi_action_request(
        self,
        normalized: str,
    ) -> bool:

        # -----------------------------------------------------
        # Explicit action separators
        # -----------------------------------------------------

        markers = (
            " and ",
            " then ",
            " after that ",
            " followed by ",
            " next ",
        )

        if any(
            marker in normalized
            for marker in markers
        ):

            return True

        # -----------------------------------------------------
        # Comma-separated commands
        #
        # Example:
        # "Open Safari, take a screenshot"
        # -----------------------------------------------------

        if "," in normalized:

            command_words = (
                "open ",
                "launch ",
                "start ",
                "close ",
                "quit ",
                "take ",
                "capture ",
                "set ",
                "mute",
                "unmute",
                "create ",
                "make ",
                "show ",
                "list ",
            )

            parts = [
                part.strip()
                for part in normalized.split(",")
                if part.strip()
            ]

            if len(parts) >= 2:

                command_like_parts = 0

                for part in parts:

                    if part.startswith(
                        command_words
                    ):

                        command_like_parts += 1

                if command_like_parts >= 2:

                    return True

        return False

    # =========================================================
    # PATH ALIASES
    # =========================================================

    def _normalize_path_alias(
        self,
        path: str,
    ) -> str:

        aliases = {
            "desktop": "~/Desktop",
            "my desktop": "~/Desktop",

            "documents": "~/Documents",
            "my documents": "~/Documents",

            "downloads": "~/Downloads",
            "my downloads": "~/Downloads",

            "home": "~",
            "home folder": "~",
            "home directory": "~",
        }

        normalized = self._normalize(
            path
        )

        return aliases.get(
            normalized,
            path,
        )

from dataclasses import dataclass, field
import inspect
import json
from typing import Any


@dataclass
class PlanStep:
    """One validated action in a RIYA plan."""

    action: str
    parameters: dict[str, Any] = field(default_factory=dict)
    description: str = ""


@dataclass
class Plan:
    """A complete plan created for a user request."""

    goal: str
    steps: list[PlanStep] = field(default_factory=list)


class AgentPlanner:
    """
    Convert natural-language requests into validated tool plans.

    The planner never executes tools. All execution remains
    under RIYA's existing ToolExecutor and security system.
    """

    def __init__(
        self,
        ai_runtime,
        tool_registry,
        semantic_engine=None,
        max_steps: int = 8,
    ):
        self.ai = ai_runtime
        self.tools = tool_registry
        self.semantic = semantic_engine
        self.max_steps = max(1, min(max_steps, 8))

    def create_plan(self, user_request: str) -> Plan:
        request = user_request.strip()

        if not request:
            return Plan(goal="", steps=[])

        tool_descriptions = self._get_tool_descriptions(request)

        prompt = f"""
You are RIYA's planning engine.

Convert the user's request into a safe sequence of tool actions.

USER REQUEST:
{request}

AVAILABLE REGISTERED TOOLS:
{tool_descriptions}

RETURN ONLY VALID JSON in this exact format:

{{
  "steps": [
    {{
      "action": "registered_tool_name",
      "parameters": {{}},
      "description": "Short explanation of this action"
    }}
  ]
}}

RULES:

1. Use only tools listed under AVAILABLE REGISTERED TOOLS.
2. Never invent a tool or parameter.
3. Use the exact parameter names provided by the tool.
4. Every action must be a registered tool.
5. Include all required parameters.
6. Break multi-action requests into ordered steps.
7. Do not create unnecessary steps.
8. Do not execute any action.
9. Never include Python, shell commands, or executable code.
10. If the request is ordinary conversation and requires no tool,
    return an empty steps array.
11. Maximum steps: {self.max_steps}.
12. Return JSON only. No Markdown or explanation.

JSON:
"""

        response = self.ai.generate(prompt)
        data = self._parse_json(response)

        if not isinstance(data, dict):
            raise ValueError("Planner response must be a JSON object.")

        raw_steps = data.get("steps", [])

        if not isinstance(raw_steps, list):
            raise ValueError("Planner steps must be a JSON array.")

        if len(raw_steps) > self.max_steps:
            raise ValueError(
                f"Planner returned too many steps. Maximum: {self.max_steps}"
            )

        validated_steps = []

        for index, raw_step in enumerate(raw_steps, start=1):
            validated_steps.append(
                self._validate_step(raw_step, index)
            )

        return Plan(
            goal=request,
            steps=validated_steps,
        )

    def _get_tool_descriptions(self, request: str) -> str:
        if self.semantic is not None:
            descriptions = self.semantic.relevant_tool_descriptions(
                request
            )

            if descriptions.strip():
                return descriptions

        lines = []

        for tool in self.tools.list_tools():
            parameters = ", ".join(sorted(tool.parameters))

            lines.append(
                f"- {tool.name}({parameters}): {tool.description}"
            )

        return "\n".join(lines)

    @staticmethod
    def _parse_json(response: str) -> dict:
        if not isinstance(response, str) or not response.strip():
            raise ValueError("RIYA returned an empty planning response.")

        start = response.find("{")

        if start == -1:
            raise ValueError(
                f"RIYA returned no JSON plan: {response}"
            )

        try:
            data, _ = json.JSONDecoder().raw_decode(response[start:])
        except json.JSONDecodeError as exc:
            raise ValueError(
                f"RIYA returned invalid planning JSON: {exc}"
            ) from exc

        return data

    def _validate_step(self, raw_step: Any, index: int) -> PlanStep:
        if not isinstance(raw_step, dict):
            raise ValueError(
                f"Plan step {index} must be a JSON object."
            )

        action = raw_step.get("action")
        parameters = raw_step.get("parameters", {})
        description = raw_step.get("description", "")

        if not isinstance(action, str) or not action.strip():
            raise ValueError(
                f"Plan step {index} has an invalid action."
            )

        action = action.strip()

        if not isinstance(parameters, dict):
            raise ValueError(
                f"Parameters for step {index} must be an object."
            )

        if not all(isinstance(key, str) for key in parameters):
            raise ValueError(
                f"Parameter names in step {index} must be strings."
            )

        if not isinstance(description, str):
            raise ValueError(
                f"Description for step {index} must be a string."
            )

        tool = self.tools.get(action)

        if tool is None:
            raise ValueError(
                f"Planner selected an unregistered tool: {action}"
            )

        # Reject parameters not explicitly allowed by the registry.
        self.tools.validate_parameters(action, parameters)

        # Check required function arguments as well.
        try:
            signature = inspect.signature(tool.function)

            required = {
                parameter.name
                for parameter in signature.parameters.values()
                if parameter.kind in (
                    inspect.Parameter.POSITIONAL_OR_KEYWORD,
                    inspect.Parameter.KEYWORD_ONLY,
                )
                and parameter.default is inspect.Parameter.empty
            }

            missing = required - set(parameters)

            if missing:
                raise ValueError(
                    f"Missing required parameters for '{action}': "
                    f"{sorted(missing)}"
                )

        except (TypeError, ValueError) as exc:
            if isinstance(exc, ValueError) and str(exc).startswith(
                "Missing required parameters"
            ):
                raise

        return PlanStep(
            action=action,
            parameters=parameters,
            description=description.strip() or action,
        )

import numpy as np

class SemanticToolEngine:
    """
    RIYA local semantic understanding engine.

    Uses a pretrained embedding model to determine which
    RIYA tools are semantically relevant to a user request.

    Important:

        Embeddings DO NOT execute tools.

    They only retrieve the tools that are most relevant.

    Execution remains controlled by:

        SecurityManager
            ↓
        ToolExecutor
            ↓
        Verification
    """

    MODEL_NAME = "sentence-transformers/all-MiniLM-L6-v2"

    def __init__(
        self,
        tool_registry,
        model_name: str = MODEL_NAME,
        top_k: int = 5,
    ):
        self.tools = tool_registry
        self.top_k = max(1, top_k)

        self.model = self._load_model(
            model_name
        )

        self._tool_texts = []
        self._tool_names = []
        self._tool_embeddings = None

        self._build_index()

    # =========================================================
    # MODEL
    # =========================================================

    def _load_model(
        self,
        model_name: str,
    ):
        """
        Load the pretrained embedding model.

        First execution may download the model.
        Later executions use the local cache.
        """

        try:
            return SentenceTransformer(
                model_name
            )

        except Exception as exc:

            raise RuntimeError(
                "RIYA could not load the local "
                "embedding model.\n\n"
                "The first run requires internet access "
                "to download the pretrained model.\n"
                "After that, RIYA can use the cached model "
                "offline.\n\n"
                f"Original error: {exc}"
            ) from exc

    # =========================================================
    # BUILD TOOL INDEX
    # =========================================================

    def _build_index(self):
        """
        Convert RIYA's registered tools into embeddings.
        """

        tools = self.tools.list_tools()

        if not tools:
            raise RuntimeError(
                "RIYA has no tools available for "
                "semantic indexing."
            )

        self._tool_texts = []
        self._tool_names = []

        for tool in tools:

            text = self._build_tool_text(
                tool
            )

            self._tool_texts.append(
                text
            )

            self._tool_names.append(
                tool.name
            )

        self._tool_embeddings = (
            self.model.encode(
                self._tool_texts,
                normalize_embeddings=True,
                convert_to_numpy=True,
            )
        )

    # =========================================================
    # TOOL DESCRIPTION
    # =========================================================

    def _build_tool_text(
        self,
        tool,
    ) -> str:
        """
        Build semantic information for a tool.

        This is NOT a list of command rules.

        It describes what the capability means.
        """

        parameters = ", ".join(
            sorted(tool.parameters)
        )

        return (
            f"Tool: {tool.name}. "
            f"Capability: {tool.description}. "
            f"Parameters: {parameters}."
        )

    # =========================================================
    # RETRIEVE
    # =========================================================

    def retrieve(
        self,
        user_request: str,
        top_k: int | None = None,
    ) -> list[SemanticMatch]:
        """
        Retrieve the most semantically relevant tools.
        """

        request = user_request.strip()

        if not request:
            return []

        if self._tool_embeddings is None:
            self._build_index()

        k = top_k or self.top_k

        k = min(
            k,
            len(self._tool_names),
        )

        query_embedding = (
            self.model.encode(
                request,
                normalize_embeddings=True,
                convert_to_numpy=True,
            )
        )

        scores = np.dot(
            self._tool_embeddings,
            query_embedding,
        )

        ranked_indices = np.argsort(
            scores
        )[::-1][:k]

        matches = []

        for index in ranked_indices:

            tool_name = (
                self._tool_names[index]
            )

            tool = self.tools.get(
                tool_name
            )

            matches.append(
                SemanticMatch(
                    tool_name=tool_name,
                    description=tool.description,
                    score=float(
                        scores[index]
                    ),
                )
            )

        return matches

    # =========================================================
    # BEST TOOL
    # =========================================================

    def best_match(
        self,
        user_request: str,
    ) -> SemanticMatch | None:

        matches = self.retrieve(
            user_request,
            top_k=1,
        )

        if not matches:
            return None

        return matches[0]

    # =========================================================
    # RELEVANT TOOL TEXT
    # =========================================================

    def relevant_tool_descriptions(
        self,
        user_request: str,
        top_k: int | None = None,
    ) -> str:
        """
        Return the most relevant tools in a format that
        Qwen3 can use for structured planning.
        """

        matches = self.retrieve(
            user_request,
            top_k=top_k,
        )

        if not matches:
            return ""

        lines = []

        for match in matches:

            tool = self.tools.get(
                match.tool_name
            )

            parameters = ", ".join(
                sorted(tool.parameters)
            )

            if parameters:

                lines.append(
                    f"- {tool.name}"
                    f"({parameters}): "
                    f"{tool.description}"
                    f" [semantic_score="
                    f"{match.score:.3f}]"
                )

            else:

                lines.append(
                    f"- {tool.name}(): "
                    f"{tool.description}"
                    f" [semantic_score="
                    f"{match.score:.3f}]"
                )

        return "\n".join(
            lines
        )

class ToolDecisionEngine:
    """Convert natural-language requests into safe tool decisions."""

    def __init__(self, ai_runtime, tool_registry):
        self.ai = ai_runtime
        self.tools = tool_registry

    def _tool_descriptions(self) -> str:
        """Build the tool list sent to the local model."""

        descriptions = []

        for tool in self.tools.list_tools():
            descriptions.append(
                f"- {tool.name}: {tool.description}"
            )

        return "\n".join(descriptions)

    def decide(self, request: str) -> ToolDecision:
        """Ask the local AI which registered tool should handle a request."""

        prompt = f"""
You are RIYA's tool decision engine.

Your job is ONLY to decide whether the user's request
should use one of the available tools.

Available tools:
{self._tool_descriptions()}

User request:
{request}

Return ONLY valid JSON in this exact format:

{{
  "tool": "tool_name_or_null",
  "parameters": {{}},
  "reason": "short explanation"
}}

Rules:
- Use only tools from the available tool list.
- Never invent a tool.
- Never invent parameters that the tool does not need.
- If no tool is appropriate, use null for "tool".
- Do not execute anything.
- Do not include markdown.
"""

        response = self.ai.generate(prompt)

        return self._parse_response(response)

    def _parse_response(self, response: str) -> ToolDecision:
        """Extract, parse, and validate the first JSON object."""

        response = response.strip()

        start = response.find("{")

        if start == -1:
            raise ValueError(
                f"RIYA returned no JSON tool decision: {response}"
            )

        depth = 0
        end = None

        for index in range(start, len(response)):
            if response[index] == "{":
                depth += 1

            elif response[index] == "}":
                depth -= 1

                if depth == 0:
                    end = index + 1
                    break

        if end is None:
            raise ValueError(
                f"RIYA returned incomplete JSON: {response}"
            )

        json_text = response[start:end]

        try:
            data = json.loads(json_text)

        except json.JSONDecodeError as exc:
            raise ValueError(
                f"RIYA returned invalid tool JSON: {json_text}"
            ) from exc

        tool_name = data.get("tool")
        parameters = data.get("parameters", {})
        reason = data.get("reason", "")

        if tool_name is not None and not self.tools.has(tool_name):
            raise ValueError(
                f"RIYA selected an unavailable tool: {tool_name}"
            )

        if not isinstance(parameters, dict):
            raise ValueError(
                "RIYA tool parameters must be an object."
            )

        # Normalize common parameter naming variations.
        if tool_name == "set_volume":
            if "level" not in parameters and "volume" in parameters:
                parameters["level"] = parameters.pop("volume")

        return ToolDecision(
            tool=tool_name,
            parameters=parameters,
            reason=reason,
        )

from core.agent.tool_registry import ToolRegistry
from core.security.manager import SecurityManager


class ToolExecutor:
    """
    Central execution layer for RIYA.

    Responsibilities:
    - Validate tools
    - Validate parameters
    - Apply security rules
    - Handle confirmation
    - Execute tools
    - Verify important actions
    - Execute multi-step plans safely
    """

    def __init__(self, registry: ToolRegistry):
        self.registry = registry
        self.security = SecurityManager()

    # =========================================================
    # SINGLE TOOL EXECUTION
    # =========================================================

    def execute(
        self,
        tool_name: str,
        confirm: bool = False,
        **parameters,
    ):
        """
        Execute one RIYA tool.

        Returns a structured result.
        """

        # -----------------------------------------------------
        # 1. Find tool
        # -----------------------------------------------------

        tool = self.registry.get(tool_name)

        if tool is None:
            raise ValueError(
                f"RIYA does not have a tool named "
                f"'{tool_name}'."
            )

        # -----------------------------------------------------
        # 2. Validate parameters
        # -----------------------------------------------------

        self.registry.validate_parameters(
            tool_name,
            parameters,
        )

        # -----------------------------------------------------
        # 3. Security check
        # -----------------------------------------------------

        decision = self.security.check(
            tool_name
        )

        if not decision.allowed:
            raise PermissionError(
                f"RIYA blocked '{tool_name}': "
                f"{decision.reason}"
            )

        # -----------------------------------------------------
        # 4. Confirmation
        # -----------------------------------------------------

        if (
            decision.requires_confirmation
            and not confirm
        ):
            return {
                "status": "confirmation_required",
                "tool": tool_name,
                "parameters": parameters,
                "reason": decision.reason,
            }

        # -----------------------------------------------------
        # 5. Execute tool
        # -----------------------------------------------------

        try:
            result = tool.function(
                **parameters
            )

        except Exception as exc:
            return {
                "status": "failed",
                "tool": tool_name,
                "parameters": parameters,
                "error": str(exc),
            }

        # -----------------------------------------------------
        # 6. Verify execution
        # -----------------------------------------------------

        verification = self.verify(
            tool_name,
            parameters,
            result,
        )

        if not verification["success"]:
            return {
                "status": "verification_failed",
                "tool": tool_name,
                "parameters": parameters,
                "result": result,
                "reason": verification["reason"],
            }

        # -----------------------------------------------------
        # 7. Successful execution
        # -----------------------------------------------------

        return {
            "status": "success",
            "tool": tool_name,
            "parameters": parameters,
            "result": result,
            "verification": verification,
        }

    # =========================================================
    # MULTI-STEP PLAN EXECUTION
    # =========================================================

    def execute_plan(
        self,
        plan,
        confirm: bool = False,
    ):
        """
        Execute a RIYA multi-step plan.

        Execution stops when:
        - confirmation is required
        - a tool fails
        - verification fails
        """

        results = []

        for step in plan.steps:

            # -------------------------------------------------
            # Internal planning step
            # -------------------------------------------------

            if step.action == "understand":

                results.append({
                    "status": "success",
                    "action": "understand",
                    "description": step.description,
                    "result": "Task understood.",
                })

                continue

            # -------------------------------------------------
            # Execute step
            # -------------------------------------------------

            result = self.execute(
                step.action,
                confirm=confirm,
                **step.parameters,
            )

            step_result = {
                "action": step.action,
                "description": step.description,
                "result": result,
            }

            results.append(step_result)

            # -------------------------------------------------
            # Confirmation required
            # -------------------------------------------------

            if (
                isinstance(result, dict)
                and result.get("status")
                == "confirmation_required"
            ):
                break

            # -------------------------------------------------
            # Execution failure
            # -------------------------------------------------

            if (
                isinstance(result, dict)
                and result.get("status")
                in {
                    "failed",
                    "verification_failed",
                }
            ):
                break

        return results

    # =========================================================
    # VERIFICATION
    # =========================================================

    def verify(
        self,
        tool_name: str,
        parameters: dict,
        result,
    ) -> dict:
        """
        Verify that an important action actually happened.
        """

        # -----------------------------------------------------
        # Open application
        # -----------------------------------------------------

        if tool_name == "open_application":

            app_name = parameters.get(
                "app_name"
            )

            if not app_name:

                    "success": False,
                    "reason": (
                        "Application name was missing."
                    ),
                }

            return self._verify_application_running(

            )

        # -----------------------------------------------------
        # Close application
        # -----------------------------------------------------

        if tool_name == "close_application":

            app_name = parameters.get(
                "app_name"
            )

            if not app_name:
                return {
                    "success": False,
                    "reason": (
                        "Application name was missing."
                    ),
                }

            return self._verify_application_closed(
                app_name
            )

        # -----------------------------------------------------
        # Screenshot
        # -----------------------------------------------------

        if tool_name == "take_screenshot":

            return self._verify_screenshot(
                result
            )

        # -----------------------------------------------------
        # Create folder
        # -----------------------------------------------------

        if tool_name == "create_folder":

            path = parameters.get(
                "path"
            )

            return self._verify_path_exists(
                path,
                expected_type="directory",
            )

        # -----------------------------------------------------
        # Write file
        # -----------------------------------------------------

        if tool_name == "write_file":

            path = parameters.get(
                "path"
            )

            return self._verify_path_exists(
                path,
                expected_type="file",
            )

        # -----------------------------------------------------
        # Delete file
        # -----------------------------------------------------

        if tool_name == "delete_file":

            path = parameters.get(
                "path"
            )

            return self._verify_path_missing(
                path
            )

        # -----------------------------------------------------
        # Delete folder
        # -----------------------------------------------------

        if tool_name == "delete_folder":

            path = parameters.get(

            )

            return self._verify_path_missing(
                path
            )

        # Other tools
        # -----------------------------------------------------

        if isinstance(result, dict):

            if result.get("status") == "failed":
                return {
                    "success": False,
                    "reason": result.get(
                        "error",
                        "Tool execution failed.",
                    ),
                }

        return {
            "success": True,
            "reason": "Tool completed successfully.",
        }

    # VERIFY SCREENSHOT
    # =========================================================

    def _verify_screenshot(
        self,
        result,
    ):
        from pathlib import Path

        if not isinstance(result, str):
            return {
                "success": False,
                "reason": (
                    "Screenshot tool did not return "
                    "a file path."
                ),
            }

        possible_path = None

        # -----------------------------------------------------
        # Try paths contained in the tool result
        # -----------------------------------------------------

        for token in result.split():

            cleaned = token.strip(
                ".,:;()[]{}'\""
            )

            candidate = Path(
                cleaned
            ).expanduser()

            if (
                cleaned.startswith("/")
                and candidate.exists()
            ):
                possible_path = candidate
                break

        # -----------------------------------------------------
        # Search Desktop for recent screenshot
        # -----------------------------------------------------

        if possible_path is None:

            desktop = (
                Path.home()
                / "Desktop"
            )

            if desktop.exists():

                screenshots = sorted(
                    desktop.glob(
                        "*screenshot*.png"
                    ),
                    key=lambda path: path.stat().st_mtime,
                    reverse=True,
                )

                if screenshots:
                    possible_path = (
                        screenshots[0]
                    )

        # -----------------------------------------------------
        # Could not verify
        # -----------------------------------------------------

        if possible_path is None:
            return {
                "success": False,
                "reason": (
                    "Screenshot was reported as created, "
                    "but RIYA could not verify the file."
                ),
            }

        # File must exist
        # -----------------------------------------------------

        if not possible_path.is_file():
            return {
                "success": False,
                "reason": (
                    f"Screenshot file does not exist: "
                    f"{possible_path}"
                ),
            }

        # -----------------------------------------------------
        # File must not be empty
        # -----------------------------------------------------

        if possible_path.stat().st_size == 0:
            return {
                "success": False,
                "reason": (
                    f"Screenshot file is empty: "
                    f"{possible_path}"
                ),
            }

        return {
            "success": True,
            "reason": (
                f"Screenshot verified: "
                f"{possible_path}"
            ),
        }

    # =========================================================

    def _verify_path_exists(
        self,

                "reason": (
                    f"Path was not created: "
                    f"{resolved}"
                ),
            }

        if expected_type == "directory":

            if not resolved.is_dir():
                return {
                    "success": False,
                    "reason": (
                        f"Expected a directory but "
                        f"found: {resolved}"
                    ),
                }

        if expected_type == "file":

            if not resolved.is_file():
                return {
                    "success": False,
                    "reason": (
                        f"Expected a file but "
                        f"found: {resolved}"
                    ),
                }

        return {
            "success": True,
            "reason": (
                f"Verified: {resolved}"
            ),
        }

@dataclass
class Tool:
    """A capability available to RIYA."""

    name: str
    description: str
    function: Callable
    parameters: set[str]


class ToolRegistry:
    """Registry of tools available to the RIYA agent."""

    def __init__(self):
        self._tools: Dict[str, Tool] = {}

    def register(
        self,
        name: str,
        description: str,
        function: Callable,
        parameters: Optional[set[str]] = None,
    ) -> None:
        """Register a tool and its accepted parameters."""

        if name in self._tools:
            raise ValueError(
                f"Tool already registered: {name}"
            )

        self._tools[name] = Tool(
            name=name,
            description=description,
            function=function,
            parameters=parameters or set(),
        )

    def get(self, name: str) -> Optional[Tool]:
        """Get a registered tool."""

        return self._tools.get(name)

    def list_tools(self) -> list[Tool]:
        """Return all registered tools."""

        return list(self._tools.values())

    def has(self, name: str) -> bool:
        """Check whether a tool exists."""

        return name in self._tools

        self,
        name: str,
        parameters: dict,

        """Validate parameters before tool execution."""

        tool = self.get(name)

        if tool is None:

from dataclasses import dataclass
from enum import Enum


class MemoryIntentType(str, Enum):

    NONE = "none"
    REMEMBER = "remember"
    RECALL = "recall"
    FORGET = "forget"
    LIST = "list"


@dataclass
class MemoryIntent:

    type: MemoryIntentType
    category: str | None = None
    key: str | None = None
    value: str | None = None


class MemoryIntentDetector:

    REMEMBER_PREFIXES = (
        "remember that ",
        "remember ",
        "save that ",
        "save ",
        "store that ",
        "store ",
        "keep in mind that ",
        "keep in mind ",
        "don't forget that ",
        "dont forget that ",
    )

    RECALL_PREFIXES = (
        "what do you remember about ",
        "what do you remember ",
        "do you remember ",
        "recall ",
        "tell me what you remember about ",
    )

    FORGET_PREFIXES = (
        "forget that ",
        "forget ",
        "remove the memory of ",
        "delete the memory of ",
    )

    LIST_PHRASES = {
        "list my memories",
        "show my memories",
        "show me my memories",
        "what do you remember about me",
        "what do you remember about me?",
        "what do you know about me",
        "what do you know about me?",
    }

    def detect(
        self,
        text: str,
    ) -> MemoryIntent:

        text = text.strip()

        if not text:
            return MemoryIntent(
                MemoryIntentType.NONE
            )

        normalized = text.lower()

        # PERSONAL MEMORY RECALL
        personal_question = normalized.rstrip(" ?.!")

        personal_recall_questions = {
            "what is my name",
            "what's my name",
            "what is my full name",
            "who am i",
            "tell me my name",
            "do you know my name",
        }

        if personal_question in personal_recall_questions:
            return MemoryIntent(
                type=MemoryIntentType.RECALL,
                category="personal",
                key="my_name",
            )

        # LIST
        if normalized in self.LIST_PHRASES:

            return MemoryIntent(
                MemoryIntentType.LIST
            )

        # REMEMBER
        for prefix in self.REMEMBER_PREFIXES:

            if normalized.startswith(prefix):

                content = text[
                    len(prefix):
                ].strip()

                if content:

                    return (
                        self._build_remember_intent(
                            content
                        )
                    )

        # FORGET
        for prefix in self.FORGET_PREFIXES:

            if normalized.startswith(prefix):

                content = text[
                    len(prefix):
                ].strip()

                if content:

                    return MemoryIntent(
                        type=(
                            MemoryIntentType.FORGET
                        ),
                        category="general",
                        key=self._make_key(
                            content
                        ),
                    )

        # RECALL
        for prefix in self.RECALL_PREFIXES:

            if normalized.startswith(prefix):

                content = text[
                    len(prefix):
                ].strip()

                if content:

                    return MemoryIntent(
                        type=(
                            MemoryIntentType.RECALL
                        ),
                        category="general",
                        key=self._make_key(
                            content
                        ),
                    )

        return MemoryIntent(
            MemoryIntentType.NONE
        )

    def _build_remember_intent(
        self,
        content: str,
    ) -> MemoryIntent:

        category = "general"

        key = self._make_key(
            content
        )

        value = content

        lowered = content.lower()

        # ------------------------------------------
        # "X is Y"
        # ------------------------------------------

        markers = (
            " is ",
            " are ",
            " = ",
        )

        for marker in markers:

            index = lowered.find(marker)

            if index > 0:

                left = content[
                    :index
                ].strip()

                right = content[
                    index + len(marker):
                ].strip()

                if left and right:

                    key = self._make_key(
                        left
                    )

                    value = right

                    category = (
                        self._infer_category(
                            left
                        )
                    )

                    return MemoryIntent(
                        type=(
                            MemoryIntentType.REMEMBER
                        ),
                        category=category,
                        key=key,
                        value=value,
                    )

        # ------------------------------------------
        # "I prefer X"
        # ------------------------------------------

        if lowered.startswith(
            "i prefer "
        ):

            value = content[
                len("I prefer "):
            ].strip()

            return MemoryIntent(
                type=(
                    MemoryIntentType.REMEMBER
                ),
                category="preferences",
                key="preferred",
                value=value,
            )

        # ------------------------------------------
        # "I like X"
        # ------------------------------------------

        if lowered.startswith(
            "i like "
        ):

            value = content[
                len("I like "):
            ].strip()

            return MemoryIntent(
                type=(
                    MemoryIntentType.REMEMBER
                ),
                category="preferences",
                key="likes",
                value=value,
            )

        # ------------------------------------------
        # "I don't like X"
        # ------------------------------------------

        if lowered.startswith(
            "i don't like "
        ):

            value = content[
                len("I don't like "):
            ].strip()

            return MemoryIntent(
                type=(
                    MemoryIntentType.REMEMBER
                ),
                category="preferences",
                key="dislikes",
                value=value,
            )

        # ------------------------------------------
        # "My favorite X is Y"
        # ------------------------------------------

        if lowered.startswith(
            "my favorite "
        ):

            marker = " is "

            index = lowered.find(
                marker
            )

            if index > 0:

                left = content[
                    len("My favorite "):
                    index
                ].strip()

                right = content[
                    index + len(marker):
                ].strip()

                if left and right:

                    return MemoryIntent(
                        type=(
                            MemoryIntentType.REMEMBER
                        ),
                        category="preferences",
                        key=(
                            "favorite_"
                            + self._make_key(
                                left
                            )
                        ),
                        value=right,
                    )

        return MemoryIntent(
            type=MemoryIntentType.REMEMBER,
            category=category,
            key=key,
            value=value,
        )

    def _infer_category(
        self,
        text: str,
    ) -> str:

        lowered = text.lower()

        if any(
            word in lowered
            for word in (
                "favorite",
                "favourite",
                "prefer",
                "preference",
                "like",
                "dislike",
            )
        ):
            return "preferences"

        if any(
            word in lowered
            for word in (
                "project",
                "application",
                "app",
                "software",
            )
        ):
            return "projects"

        if any(
            word in lowered
            for word in (
                "job",
                "work",
                "company",
                "office",
            )
        ):
            return "work"

        if any(
            word in lowered
            for word in (
                "name",
                "birthday",
                "location",
                "city",
            )
        ):
            return "personal"

        return "general"

    def _make_key(
        self,
        text: str,
    ) -> str:

        words = []

        for word in text.lower().split():

            cleaned = "".join(
                character
                for character in word
                if character.isalnum()
            )

            if cleaned:
                words.append(
                    cleaned
                )

        if not words:
            return "memory"

        return "_".join(
            words[:8]
        )

def recall_memory(
    category: str,
    key: str,
) -> str:

    value = _memory.recall(
        category=category,
        key=key,
    )

    if value is None:
        return (
            f"I don't have a memory for "
            f"'{key}' in '{category}'."
        )

    return value


def list_memories(
    category: str | None = None,
) -> str:

    memories = _memory.list_memories(
        category=category
    )

    if not memories:
        return (
            "RIYA has no stored memories."
        )

    lines = []

    for memory in memories:

        lines.append(
            f"{memory['category']}: "
            f"{memory['key']} = "
            f"{memory['value']}"
        )

    return "\n".join(lines)


def forget_memory(
    category: str,
    key: str,
) -> str:

    deleted = _memory.forget(
        category=category,
        key=key,
    )

    if not deleted:
        return (
            f"I don't have a memory for "
            f"'{key}' in '{category}'."
        )

    return (
        f"Forgot '{key}' from "
        f"'{category}'."
    )
